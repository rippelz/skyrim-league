cmake_minimum_required(VERSION 3.25)
project(RocketSkyrimBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BRIDGE_BUILD_PLUGINS "Build Windows game plugins" OFF)
include(CTest)
add_library(bridge_shared INTERFACE)
target_include_directories(bridge_shared INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(bridge_shared INTERFACE ws2_32)
endif()
if(BUILD_TESTING OR BRIDGE_BUILD_PLUGINS)
  find_package(Threads REQUIRED)
  add_executable(bridge_timing_tests tests/timing_tests.cpp)
  target_link_libraries(bridge_timing_tests PRIVATE bridge_shared Threads::Threads)
  add_test(NAME bridge_timing COMMAND bridge_timing_tests)
endif()
if(BUILD_TESTING)
  add_executable(bridge_core_tests tests/core_tests.cpp)
  target_link_libraries(bridge_core_tests PRIVATE bridge_shared)
  add_test(NAME bridge_core COMMAND bridge_core_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(WIN32)
    add_test(NAME bridge_windows_tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_windows.py")
  else()
    add_test(NAME bridge_tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*.py")
    set_tests_properties(bridge_tools PROPERTIES ENVIRONMENT "BRIDGE_CORE_BINARY=$<TARGET_FILE:bridge_core_tests>")
  endif()
endif()
if(BRIDGE_BUILD_PLUGINS)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Plugins require the x64 MSVC ABI (MSVC or clang-cl with MSVC headers/libraries). MinGW is not ABI compatible with the SDKs.")
  endif()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  add_subdirectory(bakkes-plugin)
  add_subdirectory(skse-plugin)
  add_executable(BridgeInjector tools/bridge_injector.cpp)
  target_compile_options(BridgeInjector PRIVATE /EHsc /W4)
endif()

option(BRIDGE_TEST_DYNAMIC_CONTACTS "Build the Bullet coupling regression tests" OFF)
if(BRIDGE_TEST_DYNAMIC_CONTACTS)
 if(BRIDGE_BUILD_PLUGINS)
  set(BULLET_PHYSICS_SOURCE_DIR "${BRIDGE_BULLET_ROOT}")
 endif()
 if(NOT BRIDGE_BUILD_PLUGINS)
  set(BULLET_PHYSICS_SOURCE_DIR "${PROJECT_SOURCE_DIR}/.deps/bullet-2.82")
  set(BULLET_VERSION "2.82")
  set(INSTALL_LIBS OFF)
  add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/LinearMath" "bullet-math" EXCLUDE_FROM_ALL)
  add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/BulletCollision" "bullet-collision" EXCLUDE_FROM_ALL)
  add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/BulletDynamics" "bullet-dynamics" EXCLUDE_FROM_ALL)
 endif()
  add_executable(bridge_contact_tests tests/dynamic_contact_tests.cpp bakkes-plugin/src/DynamicContacts.cpp)
  target_include_directories(bridge_contact_tests PRIVATE "${BULLET_PHYSICS_SOURCE_DIR}/src")
  target_link_libraries(bridge_contact_tests PRIVATE bridge_shared BulletDynamics BulletCollision LinearMath)
  if(MSVC)
    target_compile_definitions(bridge_contact_tests PRIVATE BT_NO_SIMD_OPERATOR_OVERLOADS)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_HOST_UNIX)
      target_compile_options(bridge_contact_tests PRIVATE /clang:-ivfsoverlay "/clang:${PROJECT_SOURCE_DIR}/.deps/bakkes-headers-vfs.json")
    endif()
  endif()
  add_test(NAME bridge_contacts COMMAND bridge_contact_tests)
endif()
