set(BAKKESMOD_SDK "${PROJECT_SOURCE_DIR}/.deps/BakkesModSDK" CACHE PATH "BakkesMod SDK root")
if(NOT EXISTS "${BAKKESMOD_SDK}/lib/pluginsdk.lib")
  message(FATAL_ERROR "Run tools/fetch_deps.py or set BAKKESMOD_SDK")
endif()
enable_language(C)
add_subdirectory("${PROJECT_SOURCE_DIR}/.deps/minhook" "${CMAKE_CURRENT_BINARY_DIR}/minhook" EXCLUDE_FROM_ALL)
set(BRIDGE_BULLET_ROOT "${PROJECT_SOURCE_DIR}/.deps/bullet-2.82" CACHE PATH "Bullet source tree with the native RL ABI")
set(BULLET_PHYSICS_SOURCE_DIR "${BRIDGE_BULLET_ROOT}")
file(READ "${BULLET_PHYSICS_SOURCE_DIR}/src/BulletCollision/CollisionShapes/btCollisionShape.h" bridge_bullet_shape_header)
if(NOT bridge_bullet_shape_header MATCHES "virtual void getAabbSlow")
  message(FATAL_ERROR "RL plugins require the uniform RL Bullet ABI. Run tools/build_windows.py / tools/build_linux.py or set BRIDGE_BULLET_ROOT to its patched source tree.")
endif()
set(BULLET_VERSION "2.82")
set(INSTALL_LIBS OFF)
add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/LinearMath" "${CMAKE_CURRENT_BINARY_DIR}/bullet-math" EXCLUDE_FROM_ALL)
add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/BulletCollision" "${CMAKE_CURRENT_BINARY_DIR}/bullet-collision" EXCLUDE_FROM_ALL)
add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/BulletDynamics" "${CMAKE_CURRENT_BINARY_DIR}/bullet-dynamics" EXCLUDE_FROM_ALL)
add_library(RocketSkyrim SHARED src/RocketSkyrim.cpp src/NativeTerrain.cpp src/DynamicContacts.cpp)
target_include_directories(RocketSkyrim PRIVATE "${BAKKESMOD_SDK}/include")
target_link_libraries(RocketSkyrim PRIVATE bridge_shared "${BAKKESMOD_SDK}/lib/pluginsdk.lib")
target_include_directories(RocketSkyrim PRIVATE "${BULLET_PHYSICS_SOURCE_DIR}/src")
target_link_libraries(RocketSkyrim PRIVATE minhook BulletDynamics BulletCollision LinearMath)
foreach(bullet_target LinearMath BulletCollision BulletDynamics RocketSkyrim)
  target_compile_definitions(${bullet_target} PRIVATE BT_NO_SIMD_OPERATOR_OVERLOADS)
endforeach()
target_compile_definitions(RocketSkyrim PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RocketSkyrim PRIVATE /EHsc /W4 /Zc:preprocessor)
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_HOST_UNIX)
  foreach(overlay_target RocketSkyrim LinearMath BulletCollision BulletDynamics)
    target_compile_options(${overlay_target} PRIVATE
      /clang:-ivfsoverlay "/clang:${PROJECT_SOURCE_DIR}/.deps/bakkes-headers-vfs.json")
  endforeach()
endif()

target_link_options(RocketSkyrim PRIVATE "/MAP:${PROJECT_SOURCE_DIR}/build-win/RocketSkyrim.map")
